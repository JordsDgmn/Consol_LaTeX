flowchart TD
    START([Start Session]) --> LOAD[Load Note]
    LOAD --> DISPLAY[Display UI]
    DISPLAY --> TIMER[Start Timer]
    TIMER --> TYPING{Typing?}
    TYPING -->|Hint| HINT[Show Hint]
    HINT --> TYPING
    TYPING -->|Continue| TIMEUP{Time Up?}
    TIMEUP -->|Yes| CAPTURE[Capture Text]
    TIMEUP -->|No| TYPING
    CAPTURE --> SIMCSE[SimCSE]
    SIMCSE --> SCORE[Score]
    SCORE --> STARS[Stars]
    STARS --> SAVE[Save]
    SAVE --> COMPLETE([Complete])

    class LOAD,SIMCSE,SCORE process
