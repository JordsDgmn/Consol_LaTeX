    subgraph "Frontend - Next.js App (localhost:3000)"

        TAILWIND[Tailwind CSS]
        COMPONENTS[UI Components<br/>• Navbar<br/>• Calendar<br/>• RadarChart<br/>• HistoryTable<br/>• LineChart]
    end
    
    subgraph "Backend - Next.js API Routes"
        NEXTAPI[Next.js API Routes<br/>• /api/users<br/>• /api/notes<br/>• /api/sessions<br/>• /api/users/upload-profile]
        NODEJS[Node.js Runtime<br/>localhost:3000]
        FORMDATA[FormData Processing<br/>Built-in Next.js<br/>File Upload Handling]
        PGPOOL[PostgreSQL Pool<br/>pg library<br/>Direct SQL Queries]
    end
    
    subgraph "Database - Local PostgreSQL"
        POSTGRES[(PostgreSQL Database<br/>localhost:5432<br/>Database: 'Consol')]
        USERS[(users table)]
        NOTES[(notes table)]
        SESSIONS[(sessions table)]
    end
    
    subgraph "File Storage - Cloud CDN"
        CLOUDINARY[Cloudinary CDN<br/>Profile Picture Storage<br/>Image Optimization]
    end
    
    subgraph "AI Processing - Separate Python Server"
        FLASK[Flask Server<br/>localhost:5000<br/>POST /score endpoint]
        SIMCSE[SimCSE Model<br/>Semantic Similarity<br/>Text Comparison Engine]
        PYTORCH[PyTorch Framework<br/>Deep Learning Runtime]
        PREPROCESSING[Text Preprocessing<br/>Markdown Cleaning<br/>Content Preparation]
        HUGGINGFACE[Hugging Face<br/>Model Download<br/>Initial Setup Only]
    end
    
    %% Frontend Flow
    BROWSER --> NEXTJS
    NEXTJS --> REACT
    REACT --> COMPONENTS
    REACT --> TAILWIND
    
    %% API Communication
    COMPONENTS -.->|HTTP Requests| NEXTAPI
    NEXTAPI --> NODEJS
    NODEJS --> PGPOOL
    
    %% Database Connection
    PGPOOL -.->|Raw SQL Queries| POSTGRES
    POSTGRES --> USERS
    POSTGRES --> NOTES
    POSTGRES --> SESSIONS
    
    %% File Upload Flow
    NEXTAPI --> FORMDATA
    FORMDATA -.->|Direct Upload| CLOUDINARY
    
    %% AI Processing Flow (Key Missing Connection!)
    NEXTAPI -.->|POST localhost:5000/score<br/>originalText + userText| FLASK
    FLASK --> PREPROCESSING
    PREPROCESSING --> SIMCSE
    SIMCSE --> PYTORCH
    FLASK -.->|similarity score<br/>0.0 to 1.0| NEXTAPI
    
    %% Model Setup (One-time)
    HUGGINGFACE -.->|Model Download<br/>Initial Setup| SIMCSE
    
    %% Data Flows
    POSTGRES -.->|Session History<br/>Note Content| COMPONENTS
    CLOUDINARY -.->|Profile Images| COMPONENTS
    
    %% Session Scoring Flow
    COMPONENTS -.->|User Input Text| NEXTAPI
    NEXTAPI -.->|Compare with Original| FLASK

%% NOTES FOR THESIS:
%% 1. Two separate servers: Next.js (localhost:3000) and Flask (localhost:5000)
%% 2. AI processing is CONNECTED via HTTP API calls, not isolated
%% 3. Text preprocessing happens in Python Flask server, not Next.js
%% 4. Multer is installed but unused - actual uploads use Next.js FormData
%% 5. Direct PostgreSQL connection via pg pool, not ORM
%% 6. Cloudinary handles image optimization and CDN serving
%% 7. SimCSE model processes cleaned text for semantic similarity scoring
