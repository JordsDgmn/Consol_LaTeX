flowchart TD
    READY([Dashboard]) --> ACTION{Action?}
    ACTION -->|Create| CREATE[Create Note]
    ACTION -->|Edit| EDIT[Edit Note]  
    ACTION -->|Select| SELECT[Select Note]
    ACTION -->|Delete| DELETE[Delete Note]
    
    CREATE --> INPUT[Input Content]
    EDIT --> MODIFY[Modify]
    INPUT --> SAVE[Save]
    MODIFY --> UPDATE[Update]
    
    SELECT --> VALID{Valid?}
    VALID -->|No| ERROR[Error]
    ERROR --> ACTION
    VALID -->|Yes| CALC[Calculate]
    
    SAVE --> NOTE_READY([Note Ready])
    UPDATE --> NOTE_READY
    CALC --> NOTE_READY
    DELETE --> ACTION
